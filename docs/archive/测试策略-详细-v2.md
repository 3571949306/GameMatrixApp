### 3.2 ADB 多设备操�?
**MultiDeviceController �?*:
```python
# tools/emulator_control.py
import subprocess
import time
from typing import Optional

class MultiDeviceController:
    """多设备控制器（支持蓝蝶模拟器�?""
    
    def __init__(self):
        self.devices = {
            'A': 'emulator-5554',
            'B': 'emulator-5556'
        }
    
    def check_devices(self):
        """检查设备连接状�?""
        result = subprocess.run(
            ['adb', 'devices'],
            capture_output=True,
            text=True
        )
        print(result.stdout)
        return 'emulator-5554' in result.stdout and 'emulator-5556' in result.stdout
    
    def install_app(self, device: str, apk_path: str):
        """安装 APK 到指定设�?""
        device_id = self.devices.get(device)
        if not device_id:
            raise ValueError(f"Unknown device: {device}")
        
        cmd = ['adb', '-s', device_id, 'install', '-r', apk_path]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return result.returncode == 0
    
    def start_activity(self, device: str, activity_name: str):
        """启动 Activity"""
        device_id = self.devices.get(device)
        cmd = [
            'adb', '-s', device_id, 'shell', 'am', 'start',
            '-n', activity_name
        ]
        result = subprocess.run(cmd, capture_output=True, text=True)
        return result.returncode == 0
    
    def tap(self, device: str, x: int, y: int):
        """点击屏幕坐标"""
        device_id = self.devices.get(device)
        cmd = ['adb', '-s', device_id, 'shell', 'input', 'tap', str(x), str(y)]
        subprocess.run(cmd, capture_output=True)
    
    def input_text(self, device: str, text: str):
        """输入文本"""
        device_id = self.devices.get(device)
        # 转义特殊字符
        text = text.replace(' ', '%s').replace("'", "\\'")
        cmd = ['adb', '-s', device_id, 'shell', 'input', 'text', text]
        subprocess.run(cmd, capture_output=True)
    
    def take_screenshot(self, device: str, output_path: str):
        """截图"""
        device_id = self.devices.get(device)
        cmd = ['adb', '-s', device_id, 'exec-out', 'screencap', '-p']
        result = subprocess.run(cmd, capture_output=True)
        with open(output_path, 'wb') as f:
            f.write(result.stdout)
    
    def wait_for_element(self, device: str, resource_id: str, timeout: int = 30):
        """等待元素出现（通过 ADB dump�?""
        device_id = self.devices.get(device)
        start_time = time.time()
        
        while time.time() - start_time < timeout:
            # Dump UI 层次结构
            cmd = ['adb', '-s', device_id, 'shell', 'uiautomator', 'dump']
            subprocess.run(cmd, capture_output=True)
            
            # 读取 dump 文件
            cmd = ['adb', '-s', device_id, 'shell', 'cat', '/sdcard/window_dump.xml']
            result = subprocess.run(cmd, capture_output=True, text=True)
            
            if resource_id in result.stdout:
                return True
            
            time.sleep(1)
        
        return False
```
